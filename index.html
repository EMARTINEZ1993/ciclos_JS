Paso 2 de 19

¿Qué es un ciclo?

Un ciclo repite instrucciones mientras se cumpla una condición. Cada repetición es una iteración.

En general, un ciclo necesita tres cosas: un valor inicial, una condición para continuar y un cambio que permita terminar.

Puedes probar los ejemplos copiándolos en la consola del navegador (F12).

Ejemplo: contar hasta 3 
for (let i = 1; i <= 3; i++) {
  console.log(i);
}

Regla importante: comprueba que la condición pueda volverse falsa; de lo contrario, crearás un ciclo infinito.

Paso 3 de 19

Ciclo for

Usa for cuando conoces la cantidad de repeticiones. La inicialización, la condición y el cambio del contador van en una sola línea.

Sumar del 1 al 5 
let suma = 0;

for (let i = 1; i <= 5; i++) {
  suma += i;
}

console.log(suma); // 15

En cada vuelta se suma i; al llegar a 5, el ciclo termina.

Paso 4 de 19

Ciclo while

Usa while cuando la cantidad de repeticiones depende de una condición. Esta se revisa antes de cada vuelta.

Duplicar hasta superar 10 
let valor = 1;

while (valor <= 10) {
  console.log(valor);
  valor *= 2; // actualiza el valor para que el ciclo termine
}

La salida es 1, 2, 4 y 8. Si la condición es falsa desde el inicio, el cuerpo no se ejecuta.

Paso 5 de 19

Ciclo do...while

Usa do...while cuando el código deba ejecutarse al menos una vez. La condición se revisa al final.

Ejecutar una vez y revisar la condición 
let numero = 0;

do {
  console.log("El menú aparece");
} while (numero !== 0);

Aunque la condición inicialmente sea falsa, el bloque se ejecuta una vez. No olvides el punto y coma después de while (...).

Paso 6 de 19

Diferencias y cuándo usar cada ciclo

for while do...while
¿Cuándo usarlo? Sabes cuántas vueltas habrá. No sabes cuántas vueltas habrá. Necesitas ejecutar al menos una vez.
¿Dónde se evalúa la condición? Al inicio de cada vuelta. Al inicio de cada vuelta. Al final de cada vuelta.
Mínimo de ejecuciones 0 0 1
Ejemplo típico Pedir 5 notas. Pedir precios hasta escribir 0. Menú de opciones, validar un dato.

En resumen: usa for para una cantidad conocida, while para repetir mientras una condición sea verdadera y do...while para ejecutar primero y comprobar después.

break termina el ciclo; continue salta a la siguiente vuelta. Por ahora, recuerda actualizar la condición para evitar ciclos infinitos.

let formTabla = document.getElementById("formTabla");
let listaTabla = document.getElementById("listaTabla");
formTabla.addEventListener("submit", generarTabla);

function generarTabla(evento) {
  evento.preventDefault();
  let numero = parseInt(document.getElementById("numeroTabla").value, 10);
  listaTabla.innerHTML = "";

  for (let i = 1; i <= 10; i++) {
    let fila = document.createElement("li");
    fila.textContent = numero + " x " + i + " = " + numero * i;
    listaTabla.appendChild(fila);
  }
}

let formAhorro = document.getElementById("formAhorro");
let resultadoAhorro = document.getElementById("resultadoAhorro");
formAhorro.addEventListener("submit", calcularAhorro);

function calcularAhorro(evento) {
  evento.preventDefault();
  let ahorro = parseFloat(document.getElementById("ahorroMensual").value);
  let valorMeta = parseFloat(document.getElementById("meta").value);
  let acumulado = 0;
  let meses = 0;

  if (ahorro <= 0 || valorMeta <= 0) {
    resultadoAhorro.textContent = "Ingresa valores mayores que cero.";
    return;
  }

  while (acumulado < valorMeta) {
    acumulado += ahorro;
    meses++;
  }
  resultadoAhorro.textContent = "Necesitas " + meses + " meses para alcanzar la meta.";
}

Ejercicio 3: lanzamiento de dado con do...while

js/ejemplo-do-while.js 
let formDado = document.getElementById("formDado");
let resultadoDado = document.getElementById("resultadoDado");
let listaDado = document.getElementById("listaDado");
formDado.addEventListener("submit", lanzarDado);

function lanzarDado(evento) {
  evento.preventDefault();
  let numeroBuscado = parseInt(document.getElementById("objetivo").value, 10);
  let dado;
  let lanzamientos = 0;
  listaDado.innerHTML = "";

  do {
    dado = Math.floor(Math.random() * 6) + 1;
    lanzamientos++;
    let fila = document.createElement("li");
    fila.textContent = "Lanzamiento " + lanzamientos + ": salió " + dado;
    listaDado.appendChild(fila);
  } while (dado !== numeroBuscado);

  resultadoDado.textContent = "Salió el número buscado en " + lanzamientos + " lanzamiento(s).";
}

La tabla usa for porque siempre tiene 10 filas; el ahorro usa while porque depende de cuándo se alcanza la meta; el dado usa do...while porque se lanza al menos una vez.

// ejercicio1.js — Sistema de compras en una tienda

// 1. Referencias a los elementos del HTML
let formulario = document.getElementById("formulario");
let mensaje = document.getElementById("mensaje");
let cantidadTexto = document.getElementById("cantidad");
let subtotalTexto = document.getElementById("subtotal");
let descuentoTexto = document.getElementById("descuento");
let totalTexto = document.getElementById("total");


// 3. Lógica
formulario.addEventListener("submit", procesarFormulario);

function procesarFormulario(evento) {
  evento.preventDefault();
  let precios = document.getElementById("precios").value.split(",");
  let subtotal = 0;
  let cantidad = 0;
  let posicion = 0;
  let precio = parseFloat(precios[posicion].trim());
  let huboInvalido = false;

  while (posicion < precios.length && precio !== 0) {
    if (Number.isNaN(precio) || precio < 0) {
      huboInvalido = true;
    } else {
      subtotal += precio;
      cantidad++;
    }

    posicion++;
    if (posicion < precios.length) {
      precio = parseFloat(precios[posicion].trim());
    }
  }

  let descuento = subtotal > 200000 ? subtotal * 0.1 : 0;
  let total = subtotal - descuento;
  cantidadTexto.textContent = cantidad;
  subtotalTexto.textContent = "$" + subtotal.toLocaleString("es-CO");
  descuentoTexto.textContent = "$" + descuento.toLocaleString("es-CO");
  totalTexto.textContent = "$" + total.toLocaleString("es-CO");

  if (cantidad === 0) {
    mensaje.textContent = "No se registraron productos.";
  } else if (subtotal > 200000) {
    mensaje.textContent = "Se aplicó un descuento del 10 %.";
  } else {
    mensaje.textContent = "Compra calculada.";
  }
  if (huboInvalido) {
    mensaje.textContent += " Se omitieron los precios inválidos o negativos.";
  }
}

// ejercicio2.js — Registro de notas de un estudiante

// 1. Referencias
let formulario = document.getElementById("formulario");
let mensaje = document.getElementById("mensaje");
let listaNotas = document.getElementById("listaNotas");
let promedioTexto = document.getElementById("promedio");
let notasAltasTexto = document.getElementById("notasAltas");
let notaMayorTexto = document.getElementById("notaMayor");
let notaMenorTexto = document.getElementById("notaMenor");


// 3. Lógica
formulario.addEventListener("submit", procesarFormulario);

function procesarFormulario(evento) {
  evento.preventDefault();
  let suma = 0;
  let cantidadValidas = 0;
  let notasAltas = 0;
  let mayor = -1;
  let menor = 6;
  listaNotas.innerHTML = "";

  for (let i = 1; i <= 5; i++) {
    let nota = parseFloat(document.getElementById("nota" + i).value);
    let fila = document.createElement("li");
    fila.textContent = "Nota " + i + ": " + nota;

    if (Number.isNaN(nota) || nota < 0 || nota > 5) {
      fila.textContent += " (inválida)";
    } else {
      suma += nota;
      cantidadValidas++;
      if (nota >= 3) {
        notasAltas++;
      }
      if (nota > mayor) {
        mayor = nota;
      }
      if (nota < menor) {
        menor = nota;
      }
    }

    listaNotas.appendChild(fila);
  }

  if (cantidadValidas === 0) {
    mensaje.textContent = "No se ingresaron notas válidas.";
    promedioTexto.textContent = "—";
    notasAltasTexto.textContent = "—";
    notaMayorTexto.textContent = "—";
    notaMenorTexto.textContent = "—";
    return;
  }

  let promedio = suma / cantidadValidas;
  promedioTexto.textContent = promedio.toFixed(2);
  notasAltasTexto.textContent = notasAltas;
  notaMayorTexto.textContent = mayor;
  notaMenorTexto.textContent = menor;
  mensaje.textContent = promedio >= 3 ? "Aprobó." : "Reprobó.";
}

  evento.preventDefault();
  let secreto = 42; // para probar; luego: Math.floor(Math.random() * 100) + 1
  let intentosEntrada = document.getElementById("intentosEntrada").value.split(",");
  let intentos = 0;
  let maxIntentos = 7;
  let puntos = 100;
  let adivino = false;
  let posicion = 0;

  while (intentos < maxIntentos && posicion < intentosEntrada.length && adivino === false) {
    let textoNumero = intentosEntrada[posicion].trim();
    let numero = parseInt(textoNumero, 10);

      if (numero === secreto) {

        fila.textContent = numero + ": correcto.";
      } else if (numero < secreto) {
        puntos -= 10;
        fila.textContent = numero + ": el secreto es mayor.";
      } else {
        puntos -= 10;
        fila.textContent = numero + ": el secreto es menor.";
      }
    }

    posicion++;
  }
